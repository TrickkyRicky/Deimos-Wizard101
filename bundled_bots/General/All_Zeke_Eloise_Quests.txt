# Loaded in before starting
block ensure_world_ready {
    times 10 {
        if any windowvisible ['WorldView'] {
            break
        }
        sleep 0.25
    }
       if not any windowvisible ['WorldView'] {
        print 'Zeke/Eloise Bot: WorldView not visible...'

        kill
    }
}

# Go to House

# Once in house, open Spiral Door
block open_world_ui {

    mass sendkey S, 0.6
    sleep 0.25

    # Wait until the Spiral Door interaction prompt is visible
    mass waitforpath ['WorldView', 'NPCRangeWin', 'imgBackground']
    sleep 0.15

    # Press X to open the Spiral Map
    mass sendkey X, 0.1
    sleep 0.8
    print 'Spiral Map should be open now.'

    # Wait for Spiral Door UI root (prevents clicking too early)
    mass waitforpath ['WorldView', '', 'messageBoxBG', 'ControlSprite']
    sleep 0.2

    # Always force to page 1
    times 10 {
        mass sendkey LEFT_ARROW, 0.03
        sleep 0.12
    }
}

block close_mount_shop_if_open {
    if windowvisible ['WorldView', 'shopGUI', 'buyWindow'] {
        print 'Mount Shop is open; closing it (failsafe).'
        any clickwindow ['WorldView', 'shopGUI', 'buyWindow', 'exit']
        sleep 0.25
    }
}
#Closes Eloise stitch UI if open
block close_stitch_shop_if_open {
     if windowvisible ['WorldView', '', 'Exit'] {
        print 'Eloise stitching menu is open; closing it (failsafe).'
        mass clickwindow ['WorldView', '', 'Exit']
        sleep 0.25
        mass clickwindow ['WorldView', '', 'Exit']
        sleep 2
    }
}




# open zeke UI
block zeke_open_ui {

    print 'At Zeke. Opening NPC services...'
   call close_mount_shop_if_open

     # Open the NPC menu reliably (and bail out if Mount Shop is all we get)
    setvar OpenedServices = False

    times 20 {
        mass sendkey X, 0.08
        sleep 0.20

        if windowvisible ['WorldView', 'NPCServicesWin'] {
            setvar OpenedServices = True
            break
        }

        # Quest missing / already completed often makes X open Mount Shop instead
        if windowvisible ['WorldView', 'shopGUI', 'buyWindow'] {
            print 'Mount Shop opened instead of quest menu (quest likely unavailable). Closing + skipping accept.'
            call close_mount_shop_if_open
            return
        }
    }

    if OpenedServices = False {
        print 'Could not open NPCServicesWin at Zeke (not in range or UI blocked). Skipping accept.'
        return
    }
}

# ACCEPTS QUEST AFTER UI OPEN
block zeke_accept_quest {
print 'Zeke UI open. Selecting Quest'

# Attempt 1: SPACEBAR (works if quest is already highlighted)
mass sendkey SPACEBAR, 0.05
sleep 0.60

# If Mount Shop opened, close it and switch (UP then SPACE)
if windowvisible ['WorldView', 'shopGUI', 'buyWindow'] {
    print 'Mount Shop opened. Closing and selecting Find the Smiths...'
    any clickwindow ['WorldView', 'shopGUI', 'buyWindow', 'exit']
    sleep 0.25

    mass sendkey UP_ARROW, 0.03
    sleep 0.12
    mass sendkey SPACEBAR, 0.05
    sleep 0.60
}


# If still in NPCServicesWin, likely no highlight yet → DOWN once then SPACE
if windowvisible ['WorldView', 'NPCServicesWin'] and not windowvisible ['WorldView', 'shopGUI', 'buyWindow'] {
    print 'Still in NPCServicesWin. Highlighting top option (DOWN) then selecting...'
    mass sendkey DOWN_ARROW, 0.03
    sleep 0.12
    mass sendkey SPACEBAR, 0.05
    sleep 0.60

    # If that still hit Mount Shop, close and switch to the quest
    if windowvisible ['WorldView', 'shopGUI', 'buyWindow'] {
        print 'Still hit Mount Shop. Closing and switching to Find the Smiths...'
        any clickwindow ['WorldView', 'shopGUI', 'buyWindow', 'exit']
        sleep 0.25

        mass sendkey UP_ARROW, 0.03
        sleep 0.12
        mass sendkey SPACEBAR, 0.05
        sleep 0.60
    }
}

    # Advance/accept dialogue
    times 25 {
        mass sendkey SPACEBAR, 0.05
        sleep 0.06
    }

    # If Mount Shop opened by accident, close it
    if windowvisible ['WorldView', 'shopGUI', 'buyWindow'] {
        any clickwindow ['WorldView', 'shopGUI', 'buyWindow', 'exit']
        sleep 0.2
    }

    # If we’re STILL in the services menu, the quest didn't open/accept.
    # Try once more (some UIs need the focus step repeated).
    if windowvisible ['WorldView', 'NPCServicesWin'] {

        print 'Still in NPCServicesWin. Retrying selection...'
        mass sendkey DOWN_ARROW, 0.06
        sleep 0.15
        mass sendkey SPACEBAR, 0.06
        sleep 0.4

        times 25 {
            mass sendkey SPACEBAR, 0.05
            sleep 0.06
        }
    }

    # Final failsafe: if menu is still open, close it so the bot can continue.
    if windowvisible ['WorldView', 'NPCServicesWin'] {
        print 'Still in NPCServicesWin; closing menu as failsafe.'
        # Your paths.py suggestion — may be Exit or Cancel depending on UI
        any clickwindow ['WorldView', 'NPCServicesWin', 'wndDialogMain', 'Exit']
        sleep 0.2
    }


    print 'Finished accept step.'
}

# TURNS IN ZEKE QUEST
block zeke_turnin_quest {

    print 'At Zeke. Turning in quest (X + SPACE spam).'

call close_mount_shop_if_open

    # Open dialogue (no services menu on turn-in)
    times 6 {
        mass sendkey X, 0.08
        sleep 0.20

        if windowvisible ['WorldView', 'shopGUI', 'buyWindow'] {
            print 'Mount Shop opened on turn-in (quest likely already done). Closing + skipping turn-in.'
            call close_mount_shop_if_open
            return
        }
    }


    # Advance/turn-in dialogue
    times 20 {
        mass sendkey SPACEBAR, 0.05
        sleep 0.08
    }

    # Failsafe close if some window is still up
    if windowvisible ['WorldView', 'shopGUI', 'buyWindow'] {
        any clickwindow ['WorldView', 'shopGUI', 'buyWindow', 'exit']
        sleep 0.2
    }
    if windowvisible ['WorldView', 'NPCServicesWin'] {
        any clickwindow ['WorldView', 'NPCServicesWin', 'wndDialogMain', 'Exit']
        sleep 0.2
    }

call close_mount_shop_if_open

    print 'Finished turn-in step.'
    sleep 0.5
}

block combat {
    if sameplace {
        if not inrange "Teleport Semi Circle 4 Player Generic" {
            sleep 0.1 
            times 10 {
                if incombat {
                    break
                } else {
                    # add the teleport in range of mob here if it is needed gamers 
                    print 'Entering Combat: Turn Auto Combat on or pick cards ' 
                    mass tp closestmob 
                    times 2 {
                        if incombat {
                            break
                        } else {
                            sleep 1
                        }
                    }
                }
            }
            if incombat {
                mass waitforcombat completion 
                sleep 1 
            }
            if any inrange "bonus" {
                times 3 {
                    entitytp mana 
                    sleep 0.6 
                    entitytp health 
                    sleep 0.6
                }
            }
        }
    }
}
