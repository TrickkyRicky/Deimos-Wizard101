###deimos_expertmode
# @name: Kalamar Raider roaming farm
# @description: Teleports to the closest Kalamar Raider, fights it, repeats. Combat config is loaded by the bot; Auto Combat must be on.
# @format: expertmode

# Priority order: Satyr when low, blade, enchanted AoE, plain AoE, single target.
loadplaystyle '?(self.health < 40%) "Satyr" @ self | any<blade> @ self | any<mod_damage> @ spell(any<damage&aoe>) & any<damage&aoe> @ aoe | any<damage&aoe> @ aoe | any<damage> @ enemy'

loop {
    mass waitforfree

    if incombat {
        mass waitforcombat completion
        sleep 1
    } else {
        # Roaming mobs move, so re-target each pass until combat starts.
        # Quoted = exact name match; unquoted would be a substring match.
        mass entitytp "EM_Kalamar_Raider_A_01"
        times 6 {
            if incombat {
                break
            } else {
                sleep 0.5
            }
        }
    }
}
